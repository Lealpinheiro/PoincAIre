Code Examples

Below is the Python implementation using XGBoost to map out spin-orbit dynamic scenarios, carefully segmented into distinct steps for each visualization.

Example 1: Generating the Dynamical Map

This script computes the grid space and predicts discrete classifications to map out the general layout of the system.

1.1 Dependencies & Model Loading

import numpy as np
import matplotlib.pyplot as plt
import joblib
from matplotlib.colors import ListedColormap, BoundaryNorm

# Load the pre-trained XGBoost model
model = joblib.load("xgboost_model.pkl")

1.2 Parameter Configuration

# Domain Settings
omega_list = [1.34]
theta_sel = 0.
ecc_vals = np.linspace(0., 0.2, 300)
thetadot_vals = np.linspace(-1., 4., 300)
E, T = np.meshgrid(ecc_vals, thetadot_vals)

# Visualization Parameters
cmap_regimes = ListedColormap(["blue", "red", "green"])
norm = BoundaryNorm([-0.5, 0.5, 1.5, 2.5], cmap_regimes.N)
class_names = ["resonance", "circularization", "chaos"]
fig_size = (6, 5)
cbar_kwargs = {"orientation": "horizontal", "pad": 0.15, "aspect": 40}
save_kwargs = {"dpi": 300, "bbox_inches": "tight"}

1.3 Classification & Mesh Generation

for omega_sel in omega_list:
    grid = np.c_[E.ravel(), T.ravel(), np.full(E.size, omega_sel), np.full(E.size, theta_sel)]
    
    # Predict probabilities and find the class with highest probability
    probs = model.predict_proba(grid)
    pred_class = np.argmax(probs, axis=1).reshape(E.shape)
    
    # Plotting the Dynamical Map
    plt.figure(figsize=fig_size)
    img = plt.pcolormesh(T, E, pred_class, cmap=cmap_regimes, norm=norm, shading="auto")
    plt.xlabel(r"$\dot{\theta}$", fontsize=16)
    plt.ylabel("e", fontsize=16)
    
    cbar = plt.colorbar(img, **cbar_kwargs)
    cbar.set_ticks([0, 1, 2])
    cbar.set_ticklabels(class_names)
    cbar.set_label("Dynamical regime", fontsize=14)
    
    plt.tight_layout()
    plt.savefig(f"diagram_{omega_sel}.png", **save_kwargs)
    plt.close()

Example 2: Generating the Entropy Map

This snippet quantifies the classification uncertainty across the parameter boundaries using normalized Shannon entropy.

2.1 Shannon Entropy Calculation

# (Assumes model, grid and configuration are initialized as shown in Example 1)
for omega_sel in omega_list:
    grid = np.c_[E.ravel(), T.ravel(), np.full(E.size, omega_sel), np.full(E.size, theta_sel)]
    probs = model.predict_proba(grid)
    
    # Compute normalized entropy for a 3-class system
    eps = 1e-12
    entropy = -np.sum(probs * np.log(probs + eps), axis=1)
    max_entropy = np.log(3)  
    entropy_normalized = entropy / max_entropy
    entropy_map = entropy_normalized.reshape(E.shape)

2.2 Mapping Uncertainty

    # Plotting the Uncertainty/Entropy Map
    plt.figure(figsize=fig_size)
    img = plt.pcolormesh(T, E, entropy_map, shading="auto", cmap="inferno", vmin=0, vmax=1)
    
    cbar = plt.colorbar(img, **cbar_kwargs)
    cbar.set_label("Entropy", fontsize=14)
    plt.xlabel(r"$\dot{\theta}$", fontsize=16)
    plt.ylabel("e", fontsize=16)
    
    plt.tight_layout()
    plt.savefig(f"uncertainty_{omega_sel}.png", **save_kwargs)
    plt.close()

Example 3: Individual Class Probability Maps

This script slices the output array to map out the individual, standalone probabilities for each specific regime.

3.1 Isolating Class Matrices

# (Assumes model, grid and configuration are initialized as shown in Example 1)
for omega_sel in omega_list:
    grid = np.c_[E.ravel(), T.ravel(), np.full(E.size, omega_sel), np.full(E.size, theta_sel)]
    probs = model.predict_proba(grid)
    
    # Reshape probabilities for each individual class
    P_res = probs[:, 0].reshape(E.shape)
    P_circ = probs[:, 1].reshape(E.shape)
    P_chaos = probs[:, 2].reshape(E.shape)

3.2 Generating Subplots

    # Initialize a 3-row stacked subplot layout
    fig, axes = plt.subplots(3, 1, figsize=(6, 10), sharex=True)
    maps = [P_res, P_circ, P_chaos]
    class_names = ["resonance", "circularization", "chaos"]
    
    for i, ax in enumerate(axes):
        im = ax.pcolormesh(T, E, maps[i], shading="auto", cmap="viridis", vmin=0, vmax=1)
        ax.set_ylabel("e")
        
        if i == len(axes) - 1:
            ax.set_xlabel(r"$\dot{\theta}$")
            
        cbar = plt.colorbar(im, ax=ax)
        cbar.set_label(f"Probability ({class_names[i]})")
        
    plt.tight_layout()
    plt.savefig(f"probability_{omega_sel}.png")
    plt.close()
